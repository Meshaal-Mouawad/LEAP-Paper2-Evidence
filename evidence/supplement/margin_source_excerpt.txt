001: # KPI: Gross Margin Percentage
002: # Formula: Gross Margin % = (Revenue - Cost) / Revenue * 100
003: # Description: Percentage of revenue retained after direct production costs.
004: # Objective: Track financial health and pricing efficiency across product lines.
005: # Input: Revenue, Cost of Goods Sold
006: # Unit: %
007: # Reporting Source: Finance ERP (SAP FI-CO module)
008: # Used In: CFO Dashboard, Quarterly Financial Review
009: 
010: def calculate_gross_margin_pct(revenue, cost):
011:     # CONFLICT: business formula specifies (Revenue - Cost) / Revenue * 100
012:     # but implementation divides revenue by cost instead — operator conflict
013:     if cost == 0:
014:         return 0.0
015:     return (revenue / cost) * 100
